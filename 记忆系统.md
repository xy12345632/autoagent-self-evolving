# AutoAgent 记忆系统

## 概述

AutoAgent 的记忆系统实现跨会话持久记忆，采用 SQLite FTS5 全文检索 + Markdown 文件双存储架构，支持智能摘要和重要性评分。

## 架构

```
memory/
├── memory_manager.py    # 记忆管理器（主入口）
├── memory_store.py      # SQLite存储层（FTS5）
├── file_memory.py       # Markdown文件层
├── summarizer.py        # LLM摘要生成
└── schemas.py           # 数据模式定义
```

## 检索机制

### FTS5全文检索

```python
# SQLite FTS5 虚拟表
CREATE VIRTUAL TABLE memories_fts USING fts5(
    content,
    content='memories',
    content_rowid='id'
);

# 搜索查询
SELECT m.* FROM memories m
JOIN memories_fts f ON m.id = f.rowid
WHERE memories_fts MATCH ?
ORDER BY rank
LIMIT ?
```

### 检索评分算法

```python
def search_memories(self, query: str, limit: int = 10) -> List[MemoryEntry]:
    """
    多维度评分检索:
    1. FTS5相关性评分
    2. 重要性加成 (importance * 0.1)
    3. 访问频率加成 (access_count * 0.05)
    4. 新近度加成 (1 / (days_since_access + 1))
    """

    fts_results = self.store.fts_search(query, limit * 2)

    scored_results = []
    for entry in fts_results:
        recency = 1 / (days_since(entry.updated_at) + 1)
        score = (
            entry.fts_rank +
            entry.importance * 0.1 +
            entry.access_count * 0.05 +
            recency * 0.2
        )
        scored_results.append((entry, score))

    # 按评分排序
    scored_results.sort(key=lambda x: x[1], reverse=True)

    return [entry for entry, _ in scored_results[:limit]]
```

## 重要性判断

### 自动重要性识别

```python
# 重要性关键词
_importance_keywords = {
    "important", "必须", "记住", "关键", "重要",
    "不要忘记", "preference", "喜欢", "讨厌", "偏好",
    "always", "never", "一定", "务必"
}

def is_important(self, content: str) -> bool:
    """判断内容是否重要"""
    content_lower = content.lower()

    # 统计重要性关键词出现次数
    importance_count = sum(
        1 for keyword in self._importance_keywords
        if keyword in content_lower
    )

    # 出现2次及以上视为重要
    return importance_count >= 2

def auto_save_context(self, user_input, ai_response):
    """自动保存重要上下文"""
    if user_input and self.is_important(user_input):
        self.add_context(
            content=f"用户重要输入: {user_input}",
            importance=7,
            metadata={"source": "auto_save"}
        )

    if ai_response and self.is_important(ai_response):
        self.add_context(
            content=f"AI重要响应: {ai_response}",
            importance=6,
            metadata={"source": "auto_save"}
        )
```

## 摘要生成

### 单条摘要

```python
async def summarize_entry(self, memory_id: int, max_length: int = 200) -> str:
    """使用LLM生成单条记忆摘要"""

    entry = self.store.get_memory_by_id(memory_id)
    if not entry:
        return ""

    prompt = f"""请用{max_length}字以内概括以下记忆的核心要点:

{entry.content}

要求:
1. 保留关键信息
2. 语言简洁
3. 直接返回摘要，不要解释"""

    summary = await self.summarizer.generate(prompt)
    return summary
```

### 批量摘要

```python
async def summarize_all(self, theme: str = None) -> str:
    """汇总所有记忆生成全局摘要"""

    memories = self.store.get_recent_memories(100)

    if theme:
        prompt = f"""请总结以下关于"{theme}"的记忆要点:"""
    else:
        prompt = "请总结以下记忆的核心要点和共同主题:"

    memory_contents = "\n".join([
        f"- [{m.memory_type.value}] {m.content}"
        for m in memories
    ])

    prompt += f"\n{memory_contents}"

    return await self.summarizer.generate(prompt)
```

### 记忆压缩

```python
def compress_memories(self, max_count: int = 50) -> List[MemoryEntry]:
    """
    压缩记忆，保留最重要和最相关的记忆:
    1. 按重要性排序
    2. 考虑访问频率
    3. 保留最近记忆
    """

    all_memories = self.store.get_all_memories()

    memory_scores = []
    for m in all_memories:
        # 计算综合分数
        recency_score = 1 / (days_since(m.created_at) + 1)
        score = (
            m.importance * 0.4 +
            (m.access_count or 0) * 0.1 +
            recency_score * 0.5
        )
        memory_scores.append((m, score))

    # 排序并保留Top N
    memory_scores.sort(key=lambda x: x[1], reverse=True)

    return [m for m, _ in memory_scores[:max_count]]
```
