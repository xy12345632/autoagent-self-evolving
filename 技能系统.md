# AutoAgent 技能系统

## 概述

AutoAgent 的技能系统是其自进化能力的核心。系统能够从解决过的问题中自动提取模式，创建可复用的程序性技能（SKILL.md格式），实现"越用越聪明"。

## 技能系统架构

```
skills/
├── skill_manager.py       # 技能管理器（主入口）
├── skill_store.py        # 技能存储（SQLite）
├── skill_loader.py       # 技能加载器
├── skill_creator.py      # 技能创建器
├── skill_evolution.py    # 技能进化引擎
├── skill_parser.py       # SKILL.md解析器
├── skill_exporter.py     # 技能导出器
└── skill_marketplace.py  # 技能市场
```

### 技能检索算法

```python
def find_relevant_skills(self, task: str) -> List[Dict]:
    """
    基于多维度评分检索相关技能:
    1. 触发词匹配 (权重3)
    2. 名称匹配 (权重2)
    3. 描述匹配 (权重1)
    4. 使用次数加成 (0.1 * count)
    5. 成功率加成 (2 * success_rate)
    """
    task_lower = task.lower()
    scored_skills = []

    for skill in all_skills:
        score = 0

        # 触发词匹配
        if any(word in skill['trigger'] for word in task_lower.split()):
            score += 3

        # 名称匹配
        if any(word in skill['name'].lower() for word in task_lower.split()):
            score += 2

        # 描述匹配
        if any(word in skill['description'].lower() for word in task_lower.split()):
            score += 1

        # 使用统计加成
        score += skill['usage_count'] * 0.1
        score += skill['success_rate'] * 2

        if score > 0:
            scored_skills.append((skill, score))

    # 返回Top 5
    return [s[0] for s in sorted(scored_skills, key=lambda x: x[1], reverse=True)[:5]]
```

## 技能自动创建

### 创建流程

```
用户问题 → 复杂度检测 → 解决方案记录 → 模式提取 → 技能生成 → 存储验证
```

### 复杂度检测

```python
# skill_creator.py

def detect_complex_task(self, task: str) -> bool:
    """
    检测任务复杂度，满足以下条件视为复杂任务:
    1. 包含多个步骤指示词
    2. 涉及多个工具使用
    3. 重复出现类似问题
    """
    complex_indicators = [
        "帮我", "如何", "怎么", "步骤",
        "教程", "解释", "实现", "创建"
    ]

    multi_step = sum(1 for w in complex_indicators if w in task) >= 2
    has_pattern = self._check_repetition(task)

    return multi_step or has_pattern
```

### 从解决方案创建技能

```python
def create_skill_from_solution(
    self,
    problem: str,
    solution: str,
    context: dict = None
) -> dict:
    """从问题和解决方案创建技能"""

    # 提取触发词
    triggers = self._extract_triggers(problem)

    # 生成技能名称
    name = self._generate_name(problem)

    # 生成描述
    description = self._generate_description(solution)

    # 确定分类
    category = context.get('category', 'general')

    skill_data = {
        'id': f"skill_{generate_id()}",
        'name': name,
        'description': description,
        'trigger': triggers,
        'action': {
            'type': 'function',
            'function': f'handle_{category}_task'
        },
        'category': category,
        'tags': context.get('tags', []),
        'version': '1.0.0',
        'usage_count': 0,
        'success_rate': 1.0
    }

    return skill_data
```

## 技能进化

### 进化机制

```python
# skill_evolution.py

class SkillEvolutionManager:
    def __init__(self, skill_store, skill_creator):
        self.store = skill_store
        self.creator = skill_creator
        self.task_history = []
        self.tool_call_count = 0

    def should_evaluate(self) -> bool:
        """每15次工具调用评估一次"""
        return self.tool_call_count >= 15

    def evaluate_and_evolve(self) -> dict:
        """
        评估和进化流程:
        1. 分析最近任务成功率
        2. 识别低效技能
        3. 合并相似技能
        4. 优化技能参数
        """
        report = {
            'evaluated_skills': 0,
            'evolved_skills': [],
            'merged_skills': [],
            'deleted_skills': []
        }

        for skill in self.store.list_skills():
            if self._should_evolve(skill):
                evolved = self._evolve_skill(skill)
                report['evolved_skills'].append(evolved)

            if self._should_merge(skill):
                merged = self._merge_similar(skill)
                report['merged_skills'].append(merged)

            report['evaluated_skills'] += 1

        return report
```

### 进化类型

| 类型 | 说明 | 触发条件 |
|------|------|----------|
| `OPTIMIZE` | 优化技能参数 | 成功率下降 |
| `MERGE` | 合并相似技能 | 触发词高度重合 |
| `SPLIT` | 拆分为多个技能 | 使用频率不均 |
| `DEPRECATE` | 标记为废弃 | 长期低成功率 |

# skill_marketplace.py

class SkillMarketplace:
    def __init__(self):
        self.listings = []
        self.categories = [
            "programming", "writing", "research",
            "analysis", "automation", "creative"
        ]

    def publish_skill(self, skill: dict, author: str, version: str) -> str:
        """发布技能到市场"""
        listing_id = generate_listing_id()

        self.listings.append({
            'id': listing_id,
            'skill': skill,
            'author': author,
            'version': version,
            'rating': 0.0,
            'install_count': 0,
            'published_at': datetime.now()
        })

        return listing_id

    def browse(self, category: str = None, sort_by: str = "rating", limit: int = 20):
        """浏览市场"""
        filtered = self.listings

        if category:
            filtered = [l for l in filtered if l['skill']['category'] == category]

        sorted_listings = sorted(
            filtered,
            key=lambda x: getattr(x, sort_by, 0),
            reverse=True
        )

        return sorted_listings[:limit]
```

# skill_exporter.py

def export_skill(self, skill: dict, path: str = None) -> str:
    """导出单个技能为SKILL.md"""

    if path is None:
        path = f"./skills/{skill['id']}/SKILL.md"

    os.makedirs(os.path.dirname(path), exist_ok=True)

    content = self._render_skill_markdown(skill)

    with open(path, 'w', encoding='utf-8') as f:
        f.write(content)

    return path

def export_all_skills(self, skills: List[dict]) -> List[str]:
    """导出所有技能"""
    paths = []
    for skill in skills:
        path = self.export_skill(skill)
        paths.append(path)

    # 生成索引文件
    self.generate_skill_index(skills)

    return paths
```

### 加载技能

```python
# skill_loader.py

def load_skill_definition(self, skill_id: str) -> Optional[dict]:
    """从SKILL.md文件加载技能"""

    # 搜索可能的路径
    paths = [
        f"./skills/{skill_id}/SKILL.md",
        f"./skills/{skill_id}.md",
        f"./skills/{skill_id}/skill.md"
    ]

    for path in paths:
        if os.path.exists(path):
            return self.parser.parse_file(path)

    return None

def discover_skills(self) -> List[dict]:
    """自动发现skills目录下的所有技能"""
    skills = []
    skills_dir = self.skills_dir or "./skills"

    for root, dirs, files in os.walk(skills_dir):
        if 'SKILL.md' in files:
            skill = self.parser.parse_file(os.path.join(root, 'SKILL.md'))
            if skill:
                skills.append(skill)

    return skills
```
